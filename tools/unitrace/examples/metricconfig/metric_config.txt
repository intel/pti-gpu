# Example metric configuration file for unitrace --metric-config.
#
#   unitrace -k --metric-config metric_config.txt <application>
#   unitrace -q --metric-config metric_config.txt <application>
#
# Generate the full list of what your device supports, with the valid parameter
# values, using:
#
#   unitrace --gen-metric-config all_metrics.txt
#
# Only one metric group can be collected per run: enable either one predefined
# group, or any number of programmable metrics (they are collected together).
# As shipped, this file collects the predefined ComputeBasic group, which every
# supported GPU has, so it runs as is on any device. The other entries are
# disabled and show the syntax; disabled entries are not checked against the
# device. Their names and valid values come from --gen-metric-config on an
# Intel(R) Arc(TM) B-Series GPU (BMG G31, device ID 0xE221); other GPUs, and
# even other boards of the same model (the available XeCores differ), can
# differ. To collect programmable metrics instead, set ComputeBasic to
# enable=false and enable the programmables, using the names and values that
# --gen-metric-config reports for your device.

predefined{
    ComputeBasic{
        # Supported sampling types: event_based,time_based
        enable=true
    }
    MemoryProfile{
        # Supported sampling types: event_based,time_based
        enable=false
    }
}

programmable{
    XVE_INST_EXECUTED_SEND_ALL{
        # Supported sampling types: event_based,time_based
        enable=false
        parameters{
            # Valid values: 0:11,16:31
            Disaggregation=2            # XeCore 2 only
            # Valid values: 1
            Normalization_average=1
        }
    }
    XVE_INST_EXECUTED_CONTROL_ALL{
        # Supported sampling types: event_based,time_based
        enable=false
        parameters{
            # Valid values: 0:11,16:31
            Disaggregation=[0:3,16:19:3]  # one metric per value: 0,1,2,3,16,19
            # Valid values: 1
            Normalization_average=1
        }
    }
}
