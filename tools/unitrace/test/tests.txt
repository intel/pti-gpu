# ============================================================================
# Unitrace test suite definition (consumed by test_unitrace.py).
# Run 'python test_unitrace.py --list' to see the resolved matrix.
#
# Each [name] block defines one test. Keys:
#
#   command  = <command line>
#       What to run (paths relative to this directory).
#       - run_test.py commands are executed once per resolved scenario,
#         with --scenario=<value> appended, after one "native" run of the
#         bare application (driver/environment sanity check).
#       - Any other command (e.g. a check script) is scenario-independent
#         and runs as-is.
#       - Leading NAME=VALUE tokens set environment variables for this
#         test only, e.g.:  command = FOO=1 run_test.py kuku GPU
#       - Placeholders: {python}   = the python interpreter
#                       {unitrace} = the unitrace binary under test
#
#   options  = <unitrace options>
#       Unitrace options shared by ALL scenarios of this test, appended to
#       every scenario before the run (keeps scenario lines down to what
#       actually varies):
#           options  = --opencl
#           scenario = -c            -->  runs "unitrace -c --opencl ..."
#           scenario = -h            -->  runs "unitrace -h --opencl ..."
#
#   platform = <OS> [<OS>...]
#       On which OS this test exists at all (Windows, Linux). A test absent
#       here never runs there, regardless of scenarios. Omitted: inherited
#       from [defaults].
#
#   scenario = <unitrace options>
#       One scenario per line; the test runs once per line.
#       - If a block has NO scenario lines, it inherits the full [defaults]
#         scenario list.
#       - If it has ANY applicable scenario lines, those REPLACE the
#         defaults completely (they do not add to them).
#
#   defaults-scenario = True | False   (case-insensitive)
#       Overrides that implicit choice explicitly:
#       - True: include the [defaults] list in addition to the block's own
#         scenario lines (common matrix + per-test extras):
#             defaults-scenario = True
#             scenario          = -k
#       - False with no scenario lines: an empty matrix - the test runs
#         exactly once, with only its 'options' (plus the native run).
#
#   skip-scenario = <unitrace options>
#       Removes scenarios from the inherited/declared list. A skip line
#       matches every scenario that CONTAINS ALL of its tokens, in any
#       order, with or without extra tokens. Examples against defaults:
#           skip-scenario = -q
#               removes "-q" and "-q --result-dir"
#           skip-scenario = --result-dir
#               removes all eight "... --result-dir" variants
#           skip-scenario = -q --result-dir
#               removes only "-q --result-dir" (both tokens required)
#
# ----------------------------------------------------------------------------
# Per-platform lines: scenario, skip-scenario and options accept an :OS
# qualifier (OS = Windows or Linux). Unqualified lines apply to every OS;
# ":OS" lines apply only there.
#
# Resolution, done independently per OS:
#   1. platform gate: if the current OS is not in 'platform', the test
#      does not run here. Done.
#   2. Collect scenario lines that apply here (unqualified + :thisOS).
#      Non-empty -> that is the scenario list. Empty -> take [defaults].
#   3. Remove everything matched by applicable skip-scenario lines
#      (unqualified + :thisOS).
#   4. Append applicable 'options' to each remaining scenario.
#
# Examples:
#
#   Different matrix per OS (own lines on both sides):
#       [kuku]
#       command          = run_test.py kuku GPU
#       scenario:Linux   = -c
#       scenario:Linux   = -q
#       scenario:Windows = -h
#     Linux runs -c and -q; Windows runs -h. Defaults unused.
#
#   Shared lines plus per-OS extras:
#       scenario         = -c        # both
#       scenario:Linux   = -q        # Linux also runs -q
#     Linux: -c, -q.  Windows: -c.
#
#   Own scenarios on one OS only:
#       scenario:Windows = -h
#     Windows runs just -h. Linux has no applicable lines -> full defaults.
#
#   Defaults everywhere, trimmed on one OS:
#       skip-scenario:Windows = -q
#     Linux: all 16 defaults. Windows: 14 (both -q variants removed).
#
#   Test exists on one OS only:
#       platform = Linux
#     Runs the resolved scenarios on Linux; never runs on Windows.
# ============================================================================
